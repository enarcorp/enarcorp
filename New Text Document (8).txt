async function getBinanceCandles(symbol, interval = "1h", limit = 200) {
    const url = `https://api.binance.com/api/v3/klines?symbol=${symbol}&interval=${interval}&limit=${limit}`;
    const response = await fetch(url);

async function get24hrTicker(symbol) {
    const url = `https://api.binance.com/api/v3/ticker/24hr?symbol=${symbol}`;

    if (!response.ok) throw new Error(`Ticker error`);
    return await response.json();
}

function parseCandles(raw) {
    return raw.map(c => ({
        time: Number(c[0]),
        open: Number(c[1]),
        high: Number(c[2]),
        low: Number(c[3]),
        close: Number(c[4]),
        volume: Number(c[5])
    }));
}

function calculateRSI(candles, period = 14) {
    if (candles.length < period + 1) return 50;
    let gains = 0, losses = 0;
    for (let i = candles.length - period; i < candles.length; i++) {
        const diff = candles[i].close - candles[i-1].close;
        if (diff > 0) gains += diff;
        else losses += Math.abs(diff);
    }
    if (losses === 0) return 100;
    const rs = gains / losses;
    return 100 - (100 / (1 + rs));
}

function calculateATR(candles, period = 14) {
    if (candles.length < period + 1) return 0;
    let trs = [];
    for (let i = 1; i < candles.length; i++) {
        const high = candles[i].high;
        const low = candles[i].low;
        const prevClose = candles[i-1].close;
        const tr = Math.max(high - low, Math.abs(high - prevClose), Math.abs(low - prevClose));
        trs.push(tr);
    }
    const recent = trs.slice(-period);
    return recent.reduce((a,b) => a + b, 0) / period;
}

function calculateVWAP(candles) {
    let pv = 0, vol = 0;
    candles.forEach(c => {
        pv += ((c.high + c.low + c.close) / 3) * c.volume;
        vol += c.volume;
    });
    return vol === 0 ? 0 : pv / vol;
}

function calculateEMA(candles, period) {
    const closes = candles.map(c => c.close);
    const k = 2 / (period + 1);
    let ema = closes[0];
    for (let i = 1; i < closes.length; i++) {
        ema = closes[i] * k + ema * (1 - k);
    }
    return ema;
}

function getSwingPoints(candles, lookback = 3) {
    const swings = [];
    for (let i = lookback; i < candles.length - lookback; i++) {
        let swingHigh = true, swingLow = true;
        for (let j = i - lookback; j <= i + lookback; j++) {
            if (j === i) continue;
            if (candles[j].high >= candles[i].high) swingHigh = false;
            if (candles[j].low <= candles[i].low) swingLow = false;
        }
        if (swingHigh) swings.push({ type: "HIGH", index: i, price: candles[i].high, time: candles[i].time });
        if (swingLow) swings.push({ type: "LOW", index: i, price: candles[i].low, time: candles[i].time });
    }
    return swings;
}

function detectMarketStructure(candles) {
    if (candles.length < 50) return { trend: "UNKNOWN", bos: false, choch: false, mss: false };
    
    const swings = getSwingPoints(candles, 3);
    if (swings.length < 6) return { trend: "SIDEWAYS", bos: false, choch: false, mss: false };
    
    const recent = swings.slice(-8);
    const highs = recent.filter(s => s.type === "HIGH");
    const lows = recent.filter(s => s.type === "LOW");
    
    if (highs.length < 2 || lows.length < 2) return { trend: "SIDEWAYS", bos: false, choch: false, mss: false };
    
    const lastHigh = highs[highs.length-1];
    const prevHigh = highs[highs.length-2];
    const lastLow = lows[lows.length-1];
    const prevLow = lows[lows.length-2];
    
    let trend = "SIDEWAYS";
    let bos = false, choch = false, mss = false;
    
    if (lastHigh.price > prevHigh.price && lastLow.price > prevLow.price) trend = "BULLISH";
    if (lastHigh.price < prevHigh.price && lastLow.price < prevLow.price) trend = "BEARISH";
    
    const current = candles[candles.length-1];
    
    if (trend === "BULLISH" && current.close > lastHigh.price) bos = true;
    if (trend === "BEARISH" && current.close < lastLow.price) bos = true;
    if (trend === "BULLISH" && current.close < lastLow.price) choch = true;
    if (trend === "BEARISH" && current.close > lastHigh.price) choch = true;
    
    // MSS (Market Structure Shift) - ????????????? CHOCH
    if (choch) mss = true;
    
    return { trend, bos, choch, mss, lastHigh: lastHigh?.price, lastLow: lastLow?.price, swings };
}

function detectLiquiditySweep(candles) {
    if (candles.length < 30) return null;
    const current = candles[candles.length-1];
    const recent = candles.slice(-25, -1);
    const highest = Math.max(...recent.map(c => c.high));
    const lowest = Math.min(...recent.map(c => c.low));
    
    if (current.high > highest && current.close < highest) 
        return { type: "?? BUY SIDE SWEEP", level: highest };
    if (current.low < lowest && current.close > lowest) 
        return { type: "?? SELL SIDE SWEEP", level: lowest };
    return null;
}

function detectFVG(candles) {
    for (let i = candles.length-5; i >= 2 && i < candles.length; i++) {
        const c1 = candles[i-2], c2 = candles[i-1], c3 = candles[i];
        if (c1.high < c3.low) return { type: "BULLISH FVG", top: c3.low, bottom: c1.high };
        if (c1.low > c3.high) return { type: "BEARISH FVG", top: c1.low, bottom: c3.high };
    }
    return null;
}

function detectOrderBlock(candles) {
    if (candles.length < 15) return null;
    for (let i = candles.length-8; i < candles.length-2; i++) {
        const current = candles[i];
        const next = candles[i+1];
        const next2 = candles[i+2];
        
        if (current.close < current.open && next.close > next.open && next2.close > next2.open) {
            return { type: "?? BULLISH OB", high: current.high, low: current.low, strength: "HIGH" };
        }
        if (current.close > current.open && next.close < next.open && next2.close < next2.open) {
            return { type: "?? BEARISH OB", high: current.high, low: current.low, strength: "HIGH" };
        }
    }
    return null;
}

function calculateConfidence(trend, rsi, sweep, fvg, ob, bos) {
    let score = 0;
    if (trend === "BULLISH" || trend === "BEARISH") score += 20;
    if (trend !== "SIDEWAYS") score += 10;
    if ((trend === "BULLISH" && rsi > 45 && rsi < 70) || (trend === "BEARISH" && rsi < 55 && rsi > 30)) score += 20;
    else if (rsi > 30 && rsi < 70) score += 10;
    if (sweep) score += 20;
    if (fvg) score += 15;
    if (ob) score += 15;
    if (bos) score += 10;
    return Math.min(score, 100);
}

function buildSignal(structure, rsi, sweep, fvg, ob) {
    if (structure.trend === "BULLISH" && rsi > 40 && sweep?.type.includes("SELL")) return "BUY";
    if (structure.trend === "BULLISH" && ob?.type.includes("BULLISH") && rsi < 65) return "BUY";
    if (structure.trend === "BEARISH" && rsi < 60 && sweep?.type.includes("BUY")) return "SELL";
    if (structure.trend === "BEARISH" && ob?.type.includes("BEARISH") && rsi > 35) return "SELL";
    if (structure.choch && structure.trend === "BULLISH") return "BUY";
    if (structure.choch && structure.trend === "BEARISH") return "SELL";
    return "NO TRADE";
}

        console.error(error);
    }
}

function connectWebSocket() {
    if (ws) ws.close();
    
    const symbolLower = currentSymbol.toLowerCase();
    const streamUrl = `wss://stream.binance.com:9443/ws/${symbolLower}@ticker`;
    
    try {
        ws = new WebSocket(streamUrl);
        
        ws.onopen = () => {
            log(`?? WebSocket connected for ${currentSymbol}`, "#4da3ff");
        };
        
        ws.onmessage = (event) => {
            if (!isRunning) return;
            try {
                const data = JSON.parse(event.data);
                if (data.c) {
                    currentPrice = parseFloat(data.c);
                    const change = parseFloat(data.P);
                    const changeElem = document.getElementById("change24");
                    if (changeElem) {
                        const color = change >= 0 ? "#00e676" : "#ff5252";
                        changeElem.innerHTML = `<span style="color:${color}">${change >= 0 ? '+' : ''}${change.toFixed(2)}%</span>`;
                    }
                    const priceElem = document.getElementById("currentPrice");
                    if (priceElem) priceElem.innerHTML = `$${currentPrice.toFixed(2)}`;
                }
            } catch(e) {}
        };
        
        ws.onerror = (error) => {
            log(`?? WebSocket error, falling back to REST`, "#ffca28");
        };
        
        ws.onclose = () => {
            if (isRunning) {
                log(`?? WebSocket closed, reconnecting in 5s...`, "#ffca28");
                setTimeout(() => { if(isRunning) connectWebSocket(); }, 5000);
            }
        };
    } catch(e) {
        log(`?? WebSocket failed: ${e.message}`, "#ff5252");
    }
}

    if (CONFIG.useWebSocket) connectWebSocket();
    
    await analyzeMarket();
    intervalId = setInterval(analyzeMarket, CONFIG.updateInterval);
}

function stopAnalysis() {
    if (!isRunning) return;
    
    isRunning = false;
    statusText.textContent = "Offline";
    statusDot.classList.remove("online");
    startBtn.textContent = "? Start Analysis";
    startBtn.classList.remove("stop");
    
    if (ws) {
        ws.close();
        ws = null;
    }
    if (intervalId) {
        clearInterval(intervalId);
        intervalId = null;
    }
    
    log("? Analysis stopped", "#ffd54f");
}
